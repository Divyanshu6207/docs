$domains = @(
    "covestrodemo3"
)
 
$cert = New-SelfSignedCertificate `
    -DnsName $domains `
    -CertStoreLocation "Cert:\LocalMachine\My" `
    -FriendlyName "Local Dev Certificate - MyApp" `
    -KeyAlgorithm RSA `
    -KeyLength 2048 `
    -HashAlgorithm SHA256 `
    -NotAfter (Get-Date).AddYears(5)
 
$thumbprint = $cert.Thumbprint
 
Export-Certificate `
    -Cert "Cert:\LocalMachine\My\$thumbprint" `
    -FilePath "$env:TEMP\myapp-dev.cer"
 
Import-Certificate `
    -FilePath "$env:TEMP\myapp-dev.cer" `
    -CertStoreLocation "Cert:\LocalMachine\Root"
 
# Minimal fix: filter by Subject instead of FriendlyName
Get-ChildItem Cert:\LocalMachine\My | Where-Object Subject -like "*covestrodemo3*"
